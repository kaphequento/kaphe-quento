using System; // ===== ISLAND SURVIVAL ===== // Mabuhay ng 15 araw hanggang dumating ang rescue! var rng = new Random(); int day = 1; int health = 100, hunger = 100, thirst = 100, energy = 100; int wood = 0, food = 0, water = 0; bool shelter = false, fire = false; const int GoalDays = 15; const int ActionsPerDay = 3; int Clamp(int v) => Math.Clamp(v, 0, 100); void Log(string msg) => Console.WriteLine(" > " + msg); void ShowStatus(int actionsLeft) { Console.WriteLine(); Console.WriteLine($"=== ARAW {day}/{GoalDays} | Natitirang aksyon: {actionsLeft} ==="); Console.WriteLine($"HP: {health} Gutom: {hunger} Uhaw: {thirst} Lakas: {energy}"); Console.WriteLine($"Gamit: Kahoy x{wood}, Pagkain x{food}, Tubig x{water}"); Console.WriteLine($"Kubol: {(shelter ? "Meron" : "Wala")} | Apoy: {(fire ? "Meron" : "Wala")}"); } void Forage() { energy = Clamp(energy - 15); int roll = rng.Next(100); if (roll < 60) { int found = rng.Next(1, 4); food += found; Log($"Nakahanap ka ng {found} prutas/ugat."); } else if (roll < 85) { Log("Wala kang nahanap."); } else { health = Clamp(health - 15); Log("Nakakain ka ng lasong halaman! -15 HP (hindi ito nadala sa bag)."); } } void ChopWood() { energy = Clamp(energy - 20); int found = rng.Next(2, 5); wood += found; Log($"Nakakuha ka ng {found} kahoy."); } void FetchWater() { energy = Clamp(energy - 10); water += 2; Log("Kumuha ka ng 2 tubig mula sa sapa."); } void Hunt() { energy = Clamp(energy - 25); if (rng.Next(100) < 45) { int meat = rng.Next(3, 6); food += meat; Log($"Matagumpay na pangangaso! +{meat} pagkain."); } else { health = Clamp(health - 10); Log("Nakatakas ang hayop at nasugatan ka. -10 HP."); } } void BuildShelter() { if (shelter) { Log("May kubol ka na."); return; } if (wood < 6) { Log("Kailangan ng 6 kahoy."); return; } wood -= 6; energy = Clamp(energy - 20); shelter = true; Log("Nakagawa ka ng kubol! Mas maganda na ang tulog mo."); } void MakeFire() { if (fire) { Log("May apoy ka na."); return; } if (wood < 3) { Log("Kailangan ng 3 kahoy."); return; } wood -= 3; fire = true; Log("Nagsindi ka ng apoy. Mainit at ligtas ang gabi."); } void Eat() { if (food <= 0) { Log("Wala kang pagkain."); return; } food--; hunger = Clamp(hunger + 25); Log("Kumain ka. +25 Gutom."); } void Drink() { if (water <= 0) { Log("Wala kang tubig."); return; } water--; thirst = Clamp(thirst + 30); Log("Uminom ka. +30 Uhaw."); } void Rest() { energy = Clamp(energy + 25); Log("Nagpahinga ka. +25 Lakas."); } void RandomEvent() { int roll = rng.Next(100); if (roll < 15) { Log("EVENT: Bumagyo! Nasira ang kubol mo." ); if (shelter) shelter = false; else { health = Clamp(health - 10); Log("Wala kang kubol, -10 HP."); } } else if (roll < 30) { if (!fire) { health = Clamp(health - 15); Log("EVENT: Inatake ka ng mababangis na hayop sa dilim! -15 HP."); } else Log("EVENT: May mga hayop sa paligid pero natakot sila sa apoy."); } else if (roll < 40) { food += 2; Log("EVENT: Nakakita ka ng nahulog na supply sa dalampasigan! +2 pagkain."); } } void EndOfDay() { Console.WriteLine(); Console.WriteLine("--- Gabi na... ---"); hunger = Clamp(hunger - 20); thirst = Clamp(thirst - 25); int recover = 30; if (shelter) recover += 20; if (fire) recover += 10; energy = Clamp(energy + recover); Log($"Natulog ka. +{recover} Lakas."); if (fire && wood > 0 && rng.Next(100) < 50) { wood--; Log("Naubos ang isang kahoy sa apoy."); } else if (fire && wood == 0 && rng.Next(100) < 50) { fire = false; Log("Namatay ang apoy dahil wala nang kahoy."); } if (hunger == 0) { health = Clamp(health - 20); Log("Gutom na gutom ka! -20 HP."); } if (thirst == 0) { health = Clamp(health - 25); Log("Dehydrated ka na! -25 HP."); } if (energy < 20) { health = Clamp(health - 10); Log("Pagod na pagod ka! -10 HP."); } RandomEvent(); day++; } Console.WriteLine("====================================="); Console.WriteLine(" ISLAND SURVIVAL (C#)"); Console.WriteLine($" Mabuhay ng {GoalDays} araw para ma-rescue!"); Console.WriteLine("====================================="); while (health > 0 && day <= GoalDays) { for (int actionsLeft = ActionsPerDay; actionsLeft > 0 && health > 0; ) { ShowStatus(actionsLeft); Console.WriteLine(); Console.WriteLine("1) Maghanap ng pagkain 2) Magputol ng kahoy 3) Kumuha ng tubig"); Console.WriteLine("4) Mangaso 5) Gumawa ng kubol (6 kahoy)"); Console.WriteLine("6) Magsindi ng apoy (3 kahoy)"); Console.WriteLine("7) Kumain (libre) 8) Uminom (libre) 9) Magpahinga"); Console.WriteLine("0) Lumabas"); Console.Write("Piliin: "); string? input = Console.ReadLine(); bool freeAction = false; bool valid = true; if (energy < 10 && input is "1" or "2" or "4" or "5") { Log("Masyadong pagod ka! Magpahinga muna."); continue; } switch (input) { case "1": Forage(); break; case "2": ChopWood(); break; case "3": FetchWater(); break; case "4": Hunt(); break; case "5": BuildShelter(); break; case "6": MakeFire(); break; case "7": Eat(); freeAction = true; break; case "8": Drink(); freeAction = true; break; case "9": Rest(); break; case "0": Console.WriteLine("Salamat sa paglalaro!"); return; default: Log("Maling pili."); valid = false; break; } if (valid && !freeAction) actionsLeft--; } if (health > 0) EndOfDay(); } Console.WriteLine(); if (health <= 0) Console.WriteLine($"GAME OVER: Namatay ka sa Araw {day}."); else Console.WriteLine("NAKALIGTAS KA! Dumating na ang rescue helicopter. YOU WIN!");
